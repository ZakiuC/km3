cmake_minimum_required(VERSION 3.16)
project(pmm_canopen LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Threads REQUIRED)
    add_executable(pmm_canopen pmm_canopen.cpp)
    target_link_libraries(pmm_canopen PRIVATE Threads::Threads)
    target_compile_options(pmm_canopen PRIVATE -Wall -Wextra -Wpedantic)
else()
    message(STATUS "Real CAN executable requires Linux SocketCAN; protocol tests are portable.")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(protocol_tests protocol_tests.cpp)
    if(MSVC)
        target_compile_options(protocol_tests PRIVATE /W4 /utf-8)
    else()
        target_compile_options(protocol_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME canopen_protocol COMMAND protocol_tests)
endif()
