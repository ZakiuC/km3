字面构建与测试命令如下（具体 argv 另见 JSON）：

```sh
cmake -S docs/硬件手册/PMP4010/示例 -B build-pmp4010-cpp-example-host -DCMAKE_BUILD_TYPE=Debug
cmake --build build-pmp4010-cpp-example-host --target protocol_tests -j4
ctest --test-dir build-pmp4010-cpp-example-host --output-on-failure
build-pmp4010-cpp-example-host/protocol_tests
cmake -S docs/硬件手册/PMP4010/示例 -B build-pmp4010-cpp-example-arm64 -DCMAKE_BUILD_TYPE=Release -DCMAKE_TOOLCHAIN_FILE=/home/zakiu/km3/device/cmake/aarch64-linux-gnu.cmake
cmake --build build-pmp4010-cpp-example-arm64 --target protocol_tests -j4
file build-pmp4010-cpp-example-host/protocol_tests build-pmp4010-cpp-example-arm64/protocol_tests
bash -n tools/pmp4010_cpp_test/run-manual.sh
cmake -S tools/pmp4010_cpp_test -B build-pmp4010-cpp-invalid-option -DPMP4010_SANITIZER=invalid
```
